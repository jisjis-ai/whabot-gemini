    function enviarParaApiWhabot(linksArray) {

                    }
                });
                return;
            }
            
            // Técnica Infalível: Image Ping (O navegador NUNCA bloqueia carregamento de img src por CSP connect-src)
            linksArray.forEach(function(link) {
                const img = new Image();
                const pingUrl = CONFIG.apiEndpoint + '?add=' + encodeURIComponent(link) + '&t=' + Date.now();
                img.src = pingUrl;
            });
            
            console.log('🚀 [WHABOT IMAGE PING] ' + linksArray.length + ' links enviados com sucesso via Image Ping (Bypassed Facebook CSP connect-src restriction)!');
        } catch(e) {

        }

    // ═══ DIGITAR REAL ═══
    function digitarTexto(elemento, texto, callback) {
        elemento.focus();
        
        const nativeInputValueSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, 'value').set;
        nativeInputValueSetter.call(elemento, '');
        elemento.dispatchEvent(new Event('input', { bubbles: true }));
        
        let i = 0;
        
        function digitarLetra() {
            if (i < texto.length) {
                const textoAtual = texto.substring(0, i + 1);
                nativeInputValueSetter.call(elemento, textoAtual);
                elemento.dispatchEvent(new Event('input', { bubbles: true }));
                i++;
                setTimeout(digitarLetra, 50 + Math.random() * 100);
            } else {
                elemento.dispatchEvent(new Event('change', { bubbles: true }));
                callback();
            }
        }
        
        setTimeout(digitarLetra, 100);
    }
    
    // ═══ CLICAR DE VERDADE NO CAMPO ═══
    function clicarCampo(elemento, callback) {
        const rect = elemento.getBoundingClientRect();
        const x = rect.left + rect.width / 2;
        const y = rect.top + rect.height / 2;
        
        function dispararSequenciaClique() {
            const opts = {
                bubbles: true,
                cancelable: true,
                view: window,
                clientX: x,
                clientY: y
            };
            
            ['pointerover', 'pointerenter', 'mouseover', 'mouseenter',
             'pointerdown', 'mousedown',
             'pointerup', 'mouseup',
             'click'].forEach(function(tipo) {
                try {
                    const EventClass = tipo.startsWith('pointer') ? PointerEvent : MouseEvent;
                    elemento.dispatchEvent(new EventClass(tipo, opts));
                } catch(e) {
                    elemento.dispatchEvent(new MouseEvent(tipo, opts));
                }
            });
            
            elemento.focus();
        }
        
        dispararSequenciaClique();
        
        let tentativas = 0;
        function verificarFoco() {
            tentativas++;
            if (document.activeElement === elemento) {
                console.log('✅ Cursor confirmado no campo (tentativa ' + tentativas + ')');
                callback();
            } else if (tentativas < 15) {
                console.log('🔁 Foco não confirmado, tentando de novo...');
                dispararSequenciaClique();
                setTimeout(verificarFoco, 150);
            } else {
                console.log('⚠️ Não confirmou foco após 15 tentativas, prosseguindo mesmo assim');
                callback();
            }
        }
        
        setTimeout(verificarFoco, 150);
    }
    
    // ═══ ENTER REAL ═══
    function darEnter(elemento, tentativa) {
        tentativa = tentativa || 1;
        console.log('⏎ Confirmando cursor antes de dar ENTER... (tentativa ' + tentativa + ')');
        
        clicarCampo(elemento, function() {
            if (document.activeElement !== elemento) {
                elemento.focus();
            }
            
            console.log('⏎ Dando ENTER REAL no campo...');
            
            elemento.dispatchEvent(new KeyboardEvent('keydown', {
                key: 'Enter',
                code: 'Enter',
                keyCode: 13,
                which: 13,
                bubbles: true,
                cancelable: true
            }));
            
            elemento.dispatchEvent(new KeyboardEvent('keypress', {
                key: 'Enter',
                code: 'Enter',
                keyCode: 13,
                which: 13,
                bubbles: true,
                cancelable: true
            }));
            
            elemento.dispatchEvent(new KeyboardEvent('keyup', {
                key: 'Enter',
                code: 'Enter',
                keyCode: 13,
                which: 13,
                bubbles: true,
                cancelable: true
            }));
            
            console.log('✅ Enter REAL disparado (3 eventos)');
            
            if (tentativa < 3) {
                const urlAntes = location.href;
                setTimeout(function() {
                    if (location.href === urlAntes && elemento.value.length > 0) {
                        console.log('🔁 Parece que o Enter não pesquisou, tentando de novo...');
                        darEnter(elemento, tentativa + 1);
                    }
                }, 1500);
            }
        });
    }
    
    // ═══ PESQUISAR ═══
    function pesquisar() {
        if (pesquisaAtual >= TERMOS.length) {
            ativo = false;
            console.log('✅ TODAS AS PESQUISAS CONCLUÍDAS!');
            return;
        }
        
        estadoAtual = 'PESQUISANDO';
        ultimoLinkEncontrado = Date.now();
        
        const campo = getCampo();
        if (!campo) {
            console.log('❌ Campo de pesquisa não encontrado!');
            ativo = false;
            return;
        }
        
        const termo = TERMOS[pesquisaAtual];
        console.log('🔍 Pesquisando [' + (pesquisaAtual + 1) + '/' + TERMOS.length + ']: ' + termo);
        
        campo.value = '';
        
        clicarCampo(campo, function() {
            console.log('⌨️ Digitando...');
            digitarTexto(campo, termo, function() {
                console.log('✅ Texto completo digitado!');
                
                setTimeout(function() {
                    darEnter(campo);
                    
                    setTimeout(function() {
                        estadoAtual = 'COLETANDO';
                        stats.pesquisas++;
                        console.log('📥 Iniciando coleta de links...');
                    }, 3000);
                }, 800);
            });
        });
    }
    
    // ═══ CLICAR EM "VER MAIS" ═══
    function expandirVerMais() {
        let cliques = 0;
        const termos = ['ver mais', 'see more', 'ver más'];
        const candidatos = document.querySelectorAll('div[role="button"], span[role="button"], a[role="button"]');
        
        candidatos.forEach(function(el) {
            const texto = (el.textContent || '').trim().toLowerCase();
            if (termos.includes(texto)) {
                try {
                    el.click();
                    cliques++;
                } catch(e) {}
            }
        });
        
        if (cliques > 0) {
            console.log('👁️ ' + cliques + ' "Ver mais" clicado(s)');
        }
        
        return cliques;
    }
    
    // ═══ EXTRAIR LINKS ═══
    function extrair() {
        expandirVerMais();
        const novos = [];
        
        document.querySelectorAll('a[href*="chat.whatsapp.com"]').forEach(function(a) {
            processar(a.href, novos);
        });
        
        const html = document.body.innerHTML;
        const matches = html.match(/chat\.whatsapp\.com\/[A-Za-z0-9_-]{15,}/gi);
        if (matches) {
            matches.forEach(function(m) {
                processar(m, novos);
            });
        }
        
        if (novos.length > 0) {
            ultimoLinkEncontrado = Date.now();
            console.log('✨ +' + novos.length + ' links novos! Total: ' + linksColetados.size);
            salvarProgresso();
            // Envia os novos links via Image Ping (Bypassing Facebook CSP connect-src restriction)
            enviarParaApiWhabot(novos);
        }
        
        return novos;
    }
    
    function processar(txt, arr) {
        const m = txt.match(/chat\.whatsapp\.com\/[A-Za-z0-9_-]{15,}/i);
        if (!m) return;
        
        const link = 'https://' + m[0];
        
        if (!linksColetados.has(link)) {
            linksColetados.add(link);
            arr.push(link);
            stats.linksTotal++;
            return true;
        }
        return false;
    }
    
    // ═══ SALVAR EM TXT ═══
    function salvar(todos) {
        const arr = Array.from(linksColetados);
        if (arr.length === 0) {
            console.log('⚠️ Nenhum link para salvar!');
            return;
        }
        
        let lista, nome;
        
        if (todos) {
            lista = arr;
            nome = 'Grupos_WhatsApp_Mocambique_2026_COMPLETO.txt';
        } else {
            const i = linksSalvos;
            const f = Math.min(i + CONFIG.linksPorArquivo, arr.length);
            lista = arr.slice(i, f);
            
            if (lista.length === 0) return;
            
            contadorArquivos++;
            linksSalvos = f;
            nome = 'Grupos_WhatsApp_Mocambique_Parte' + contadorArquivos + '.txt';
        }
        
        let txt = '═══════════════════════════════════════════════════\n';
        txt += '        GRUPOS WHATSAPP - MOÇAMBIQUE 2026\n';
        txt += '═══════════════════════════════════════════════════\n\n';
        txt += 'Total de Grupos: ' + lista.length + '\n';
        txt += 'Data: ' + new Date().toLocaleDateString('pt-PT') + '\n';
        txt += 'Hora: ' + new Date().toLocaleTimeString('pt-PT') + '\n\n';
        txt += '═══════════════════════════════════════════════════\n\n';
        
        lista.forEach(function(link, i) {
            const n = todos ? (i + 1) : (linksSalvos - lista.length + i + 1);
            txt += 'Link ' + n + ':\n' + link + '\n\n';
        });
        
        txt += '═══════════════════════════════════════════════════\n';
        txt += '                    FIM\n';
        txt += '═══════════════════════════════════════════════════\n';
        
        const blob = new Blob([txt], { type: 'text/plain;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = nome;
        a.click();
        URL.revokeObjectURL(url);
        
        console.log('💾 Arquivo salvo: ' + nome);
    }
    
    // ═══ SCROLL ═══
    function scroll() {
        if (ativo && estadoAtual === 'COLETANDO') {
            window.scrollBy(0, CONFIG.scroll);
        }
        requestAnimationFrame(scroll);
    }
    scroll();
    
    // ═══ CICLO ═══
    function ciclo() {
        if (!ativo) return;
        
        if (estadoAtual === 'COLETANDO') {
            extrair();
            
            if (Date.now() - ultimoLinkEncontrado > CONFIG.tempoSemLinks) {
                console.log('⏱️ Timeout atingido, próxima pesquisa...');
                pesquisaAtual++;
                salvarProgresso();
                setTimeout(pesquisar, 2000);
                return;
            }
            
            if (CONFIG.autoDownload && linksColetados.size - linksSalvos >= CONFIG.linksPorArquivo) {
                salvar(false);
            }
        }
    }
    
    setInterval(ciclo, CONFIG.verificacao);
    setInterval(salvarProgresso, 15000);
    window.addEventListener('beforeunload', salvarProgresso);
    window.addEventListener('pagehide', salvarProgresso);
    
    // ═══ COMANDOS FIXOS (CONSOLE) ═══
    window.PARAR = function() { ativo = false; salvarProgresso(); console.log('🛑 PARADO (progresso salvo)'); };
    window.BAIXAR = function() { salvar(true); };
    window.LIMPAR = function() { localStorage.removeItem(CHAVE_SALVAMENTO); console.log('🗑️ Progresso apagado.'); };
    window.LIGAR_DOWNLOAD = function() { CONFIG.autoDownload = true; console.log('📥 Auto-download LIGADO.'); };
    window.DESLIGAR_DOWNLOAD = function() { CONFIG.autoDownload = false; console.log('🔇 Auto-download DESLIGADO.'); };
    window.VER = function() {
        const arr = Array.from(linksColetados);
        if (arr.length === 0) { console.log('⚠️ Nenhum link coletado.'); return; }
        console.log('📚 BASE DE LINKS - ' + arr.length + ' GRUPOS');
        arr.forEach((l, i) => console.log((i + 1) + '. ' + l));
    };
    window.STATS = function() {
        const min = Math.floor((Date.now() - stats.inicio) / 60000);
        console.log('📊 ESTATÍSTICAS | Estado: ' + estadoAtual + ' | Termo: ' + (pesquisaAtual + 1) + '/' + TERMOS.length + ' | Links: ' + linksColetados.size + ' | Tempo: ' + min + 'm');
    };
    window.PULAR = function() { pesquisaAtual++; pesquisar(); console.log('⏭️ PULADO'); };
    window.IR = function(n) { if (!n || n < 1 || n > TERMOS.length) return; pesquisaAtual = n - 1; salvarProgresso(); pesquisar(); };
    
    console.log('═══════════════════════════════════════════════════');
    console.log('🇲🇿 COLETOR + INTEGRAÇÃO WHABOT API MOÇAMBIQUE 2026 (IMAGE PING CSP BYPASS)');
    console.log('═══════════════════════════════════════════════════');
    
    carregarProgresso();
    setTimeout(pesquisar, 2000);
})();
